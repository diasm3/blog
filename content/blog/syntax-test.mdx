# Functions
function greet() {
  local name=$1
  echo "Hello, $name!"
}

# Conditional statements
if [ -f "config.txt" ]; then
  echo "Config file exists"
else
  echo "Creating config file..."
  touch config.txt
fi

# Loops
for i in {1..5}; do
  echo "Iteration $i"
done

# While loop

while [ $count -lt 3 ]; do
  echo "Count: $count"
  ((count++))
