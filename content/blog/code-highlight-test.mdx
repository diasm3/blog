# 코드 하이라이팅 테스트

이 페이지는 `rehype-pretty-code`를 사용한 코드 하이라이팅 기능을 테스트합니다.

## JavaScript 예제

```js
// 기본 함수
function greet(name) {
  console.log(`Hello, ${name}!`)
  return `Hello, ${name}!`
}

// 화살표 함수
const multiply = (a, b) => a * b

// 클래스 예제
class Person {
  constructor(name, age) {
    this.name = name
    this.age = age
  }

  sayHello() {
    console.log(
      `Hello, my name is ${this.name} and I am ${this.age} years old.`
    )
  }
}
```

## TypeScript 예제

```ts
// TypeScript 인터페이스
interface User {
  id: number
  name: string
  email: string
  isActive: boolean
}

// 타입 확장
type AdminUser = User & {
  permissions: string[]
  role: "admin" | "superadmin"
}

// 제네릭 함수
function createUser<T extends User>(user: T): T {
  return user
}

// 비동기 함수
const getUser = async (id: number): Promise<User | null> => {
  try {
    // API에서 사용자 정보 가져오기
    return { id, name: "John", email: "john@example.com", isActive: true }
  } catch (error) {
    console.error(error)
    return null
  }
}
```
